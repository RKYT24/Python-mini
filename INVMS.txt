import tkinter as tk
from tkinter import ttk, filedialog, messagebox

import sqlite3
import csv
import re

class InvMS:

    def __init__(self, root):

        self.root = root
        self.root.title("InvMS - IOL Inventory Management System.")
        self.root.geometry("1350x700")
        
        self.conn = sqlite3.connect("dynamic_data.db")             # Database
        self.cursor = self.conn.cursor()

        self.columns = []                       # Dynamic Variables
        self.entries = {}
        self.selected_id = None
        
        self.create_ui()

    # ========================= UI ============================= #

    def create_ui(self):

        # TOP FRAME
        top_frame = tk.Frame(self.root)
        top_frame.pack(fill="x", padx=10, pady=10)

        tk.Button(
            top_frame,
            text="Import CSV",
            width=15,
            command=self.import_csv
        ).pack(side="left", padx=5)

        tk.Button(
            top_frame,
            text="Export CSV",
            width=15,
            command=self.export_csv
        ).pack(side="left", padx=5)

        self.search_entry = tk.Entry(
            top_frame,
            width=30
        )
        self.search_entry.pack(side="right", padx=5)

        tk.Button(
            top_frame,
            text="Search",
            width=15,
            command=self.search_data
        ).pack(side="right", padx=5)

        # FORM FRAME
        self.form_frame = tk.Frame(self.root)
        self.form_frame.pack(
            fill="x",
            padx=10,
            pady=10
        )

        # BUTTON FRAME
        button_frame = tk.Frame(self.root)
        button_frame.pack(
            fill="x",
            padx=10,
            pady=10
        )

        tk.Button(
            button_frame,
            text="Add",
            width=15,
            bg="green",
            fg="white",
            command=self.add_record
        ).pack(side="left", padx=5)

        tk.Button(
            button_frame,
            text="Update",
            width=15,
            bg="blue",
            fg="white",
            command=self.update_record
        ).pack(side="left", padx=5)

        tk.Button(
            button_frame,
            text="Delete",
            width=15,
            bg="red",
            fg="white",
            command=self.delete_record
        ).pack(side="left", padx=5)

        tk.Button(
            button_frame,
            text="Refresh",
            width=15,
            command=self.load_table
        ).pack(side="left", padx=5)

        # TABLE FRAME

        self.table_frame = tk.Frame(self.root)
        self.table_frame.pack(
            fill="both",
            expand=True,
            padx=10,
            pady=10
        )


    # SANITIZE COLUMN NAME

    def sanitize_column(self, column):

        column = column.strip()

        # Replace spaces/symbols
        column = re.sub(r'[^a-zA-Z0-9_]', '_', column)

        # Prevent starting with number
        if column and column[0].isdigit():
            column = f"col_{column}"

        # SQLite reserved keywords
        reserved = {
            "index", "group", "table",
            "select", "where", "insert",
            "update", "delete", "order"
        }

        if column.lower() in reserved:
            column = f"col_{column}"

        return column


    # IMPORT CSV

    def import_csv(self):
        file_path = filedialog.askopenfilename(
            filetypes=[("CSV Files", "*.csv")]
        )
        if not file_path:
            return

        try:
            with open(
                file_path,
                mode="r",
                newline="",
                encoding="utf-8"
            ) as file:

                reader = csv.DictReader(file)

                if not reader.fieldnames:
                    messagebox.showerror(
                        "Error",
                        "CSV file has no headers."
                    )
                    return

                # Original Headers
                original_headers = reader.fieldnames

                # Safe SQL Headers
                self.columns = [
                    self.sanitize_column(col)
                    for col in original_headers
                ]

                # Mapping
                self.column_map = dict(
                    zip(original_headers, self.columns)
                )

                # Create Database
                self.create_table()
                # Create GUI
                self.create_form()
                self.create_treeview()
                # Insert Data
                for row in reader:
                    values = [
                        row[header]
                        for header in original_headers
                    ]
                    placeholders = ",".join(
                        ["?"] * len(values)
                    )

                    columns_sql = ",".join(
                        f'"{col}"'
                        for col in self.columns
                    )

                    query = f"""
                        INSERT INTO inventory
                        ({columns_sql})
                        VALUES ({placeholders})
                    """

                    self.cursor.execute(
                        query,
                        values
                    )

                self.conn.commit()
                self.load_table()
                messagebox.showinfo(
                    "Success",
                    "CSV imported successfully."
                )

        except Exception as e:
            messagebox.showerror(
                "Error",
                str(e)
            )


    # CREATE SQLITE TABLE
    def create_table(self):

        self.cursor.execute(
            "DROP TABLE IF EXISTS inventory"
        )

        column_sql = ", ".join(
            f'"{col}" TEXT'
            for col in self.columns
        )

        query = f"""
            CREATE TABLE inventory (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                {column_sql}
            )
        """

        self.cursor.execute(query)
        self.conn.commit()

    # CREATE FORM
    def create_form(self):
        for widget in self.form_frame.winfo_children():
            widget.destroy()
        self.entries.clear()

        for i, column in enumerate(self.columns):
            tk.Label(
                self.form_frame,
                text=column
            ).grid(
                row=0,
                column=i,
                padx=5,
                pady=5
            )
            entry = tk.Entry(
                self.form_frame,
                width=18
            )
            entry.grid(
                row=1,
                column=i,
                padx=5,
                pady=5
            )
            self.entries[column] = entry


    # CREATE TREEVIEW

    def create_treeview(self):
        for widget in self.table_frame.winfo_children():
            widget.destroy()
        tree_columns = ["id"] + self.columns

        self.tree = ttk.Treeview(
            self.table_frame,
            columns=tree_columns,
            show="headings"
        )

        for column in tree_columns:
            self.tree.heading(
                column,
                text=column
            )
            self.tree.column(
                column,
                width=150
            )

        scrollbar = ttk.Scrollbar(
            self.table_frame,
            orient="vertical",
            command=self.tree.yview
        )

        self.tree.configure(
            yscrollcommand=scrollbar.set
        )

        scrollbar.pack(
            side="right",
            fill="y"
        )

        self.tree.pack(
            fill="both",
            expand=True
        )

        self.tree.bind(
            "<<TreeviewSelect>>",
            self.select_row
        )


    # LOAD TABLE

    def load_table(self):
        if not self.columns:
            return
        for item in self.tree.get_children():
            self.tree.delete(item)
        self.cursor.execute(
            "SELECT * FROM inventory"
        )
        rows = self.cursor.fetchall()
        for row in rows:
            self.tree.insert(
                "",
                "end",
                values=row
            )

    # ADD RECORD
    def add_record(self):
        values = [
            self.entries[col].get()
            for col in self.columns
        ]
        placeholders = ",".join(
            ["?"] * len(values)
        )
        columns_sql = ",".join(
            f'"{col}"'
            for col in self.columns
        )
        query = f"""
            INSERT INTO inventory
            ({columns_sql})
            VALUES ({placeholders})
        """
        self.cursor.execute(
            query,
            values
        )
        self.conn.commit()
        self.load_table()
        self.clear_fields()

    # UPDATE RECORD

    def update_record(self):
        if self.selected_id is None:
            return
        values = [
            self.entries[col].get()
            for col in self.columns
        ]
        set_clause = ", ".join(
            f'"{col}"=?'
            for col in self.columns
        )
        query = f"""
            UPDATE inventory
            SET {set_clause}
            WHERE id=?
        """
        values.append(self.selected_id)

        self.cursor.execute(
            query,
            values
        )
        self.conn.commit()
        self.load_table()
        self.clear_fields()


    # DELETE RECORD

    def delete_record(self):
        if self.selected_id is None:
            return
        confirm = messagebox.askyesno(
            "Confirm",
            "Delete selected record?"
        )
        if not confirm:
            return
        self.cursor.execute(
            "DELETE FROM inventory WHERE id=?",
            (self.selected_id,)
        )
        self.conn.commit()
        self.load_table()
        self.clear_fields()


    # SELECT ROW

    def select_row(self, event):
        selected = self.tree.focus()
        if not selected:
            return
        values = self.tree.item(
            selected,
            "values"
        )

        if not values:
            return

        self.selected_id = values[0]

        for i, column in enumerate(self.columns):

            self.entries[column].delete(
                0,
                tk.END
            )

            self.entries[column].insert(
                0,
                values[i + 1]
            )


    # SEARCH

    def search_data(self):
        keyword_text = self.search_entry.get().strip()
        # Empty Search = Load All
        if keyword_text == "":
            self.load_table()
            return
        # Split Keywords
        keywords = keyword_text.split()
        # Clear Table
        for item in self.tree.get_children():
            self.tree.delete(item)

        # Build Query
        query_conditions = []
        values = []
        for keyword in keywords:
            column_conditions = []
            for col in self.columns:
                column_conditions.append(
                    f'"{col}" LIKE ?'
                )
                values.append(
                    f"%{keyword}%"
                )
            query_conditions.append(
                "(" + " OR ".join(column_conditions) + ")"
            )
        final_query = f"""
            SELECT * FROM inventory
            WHERE {' AND '.join(query_conditions)}
        """
        self.cursor.execute(
            final_query,
            values
        )

        rows = self.cursor.fetchall()

        # Display Results
        for row in rows:

            self.tree.insert(
                "",
                "end",
                values=row
            )

    # EXPORT CSV

    def export_csv(self):
        if not self.columns:
            return

        file_path = filedialog.asksaveasfilename(
            defaultextension=".csv",
            filetypes=[("CSV Files", "*.csv")]
        )

        if not file_path:
            return

        try:
            self.cursor.execute(
                "SELECT * FROM inventory"
            )

            rows = self.cursor.fetchall()

            with open(
                file_path,
                mode="w",
                newline="",
                encoding="utf-8"
            ) as file:
                writer = csv.writer(file)

                writer.writerow(
                    ["id"] + self.columns
                )

                writer.writerows(rows)

            messagebox.showinfo(
                "Success",
                "CSV exported successfully."
            )

        except Exception as e:
            messagebox.showerror(
                "Error",
                str(e)
            )


    # CLEAR FIELDS

    def clear_fields(self):
        self.selected_id = None
        for entry in self.entries.values():
            entry.delete(0, tk.END)

# RUN APP

if __name__ == "__main__":
    root = tk.Tk()
    app = InvMS(root)
    root.mainloop()
